% texforge example — fonts + emoji showcase.
% Requires the bundled fonts installed by name:  texforge fonts --install
% Renders with: texforge render examples/showcase/main.tex
\documentclass[12pt,a4paper]{article}

\usepackage[margin=2.2cm]{geometry}
\usepackage{fontspec}      % XeTeX font machinery (tectonic is XeTeX-based)
\usepackage{xcolor}
\usepackage{microtype}
\usepackage{parskip}

% --- Fonts, referenced by family NAME (installed via `texforge fonts --install`).
\setmainfont{Fira Sans}
\setmonofont{IBM Plex Mono}
\newfontfamily\serif{IBM Plex Serif}

% --- Emoji as a dedicated face. On tectonic/XeTeX, emoji render as crisp
%     MONOCHROME glyphs (OpenMoji). Colour bitmap/COLR emoji are not supported
%     by the XeTeX engine — this is a documented engine limitation.
\newfontface\emojifont{OpenMoji Black}
\newcommand{\e}[1]{{\emojifont\char"#1}}

\definecolor{accent}{HTML}{C1121F}
